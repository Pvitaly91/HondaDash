cmake_minimum_required(VERSION 3.25)
project(HondaDash VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(HONDADASH_BUILD_GUI "Build Qt Widgets dashboard" ON)
option(HONDADASH_WITH_SERIAL "Build Qt SerialPort transport" ON)
option(HONDADASH_BUILD_BRIDGE_CHECK "Build explicit bridge acceptance console" ${HONDADASH_BUILD_GUI})
include(CTest)
find_package(Threads REQUIRED)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    OUTPUT_VARIABLE HONDADASH_DESKTOP_SHA OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT HONDADASH_DESKTOP_SHA)
    set(HONDADASH_DESKTOP_SHA "unknown")
endif()
add_library(hondadash_core src/core/model.cpp src/protocol/protocol.cpp src/protocol/device_extension.cpp src/emulator/emulator.cpp src/application/session.cpp src/transport/in_memory_transport.cpp)
target_include_directories(hondadash_core PUBLIC src)
add_library(hondadash_dlc src/honda_dlc/protocol.cpp src/honda_dlc/profile.cpp src/honda_dlc/responder.cpp src/honda_dlc/session.cpp)
target_link_libraries(hondadash_dlc PUBLIC hondadash_core)
add_library(hondadash_recording src/recording/recorder.cpp)
target_link_libraries(hondadash_recording PUBLIC hondadash_core Threads::Threads)
add_library(nano_endpoint firmware/nano_synthetic/endpoint.cpp)
set_target_properties(nano_endpoint PROPERTIES CXX_STANDARD 11)
target_include_directories(nano_endpoint PUBLIC firmware/nano_synthetic)
add_library(nano_bridge firmware/nano_dlc_bridge_lab/endpoint.cpp firmware/nano_dlc_bridge_lab/transaction_engine.cpp firmware/nano_dlc_bridge_lab/virtual_ecu.cpp)
set_target_properties(nano_bridge PROPERTIES CXX_STANDARD 11)
target_include_directories(nano_bridge PUBLIC firmware/nano_dlc_bridge_lab)
add_library(hondadash_one_wire firmware/shared/one_wire.cpp)
set_target_properties(hondadash_one_wire PROPERTIES CXX_STANDARD 11)
target_include_directories(hondadash_one_wire PUBLIC firmware/shared)
add_library(nano_bench firmware/nano_dlc_responder_bench/endpoint.cpp)
set_target_properties(nano_bench PROPERTIES CXX_STANDARD 11)
target_link_libraries(nano_bench PUBLIC nano_bridge hondadash_one_wire)
add_library(hondadash_bridge src/bridge/client.cpp src/bridge/native_transport.cpp)
target_link_libraries(hondadash_bridge PUBLIC hondadash_dlc nano_bridge)
add_library(hondadash_bench src/bench/controller.cpp)
target_link_libraries(hondadash_bench PUBLIC hondadash_bridge)
add_library(hondadash_acceptance src/acceptance/runner.cpp)
target_link_libraries(hondadash_acceptance PUBLIC hondadash_bench hondadash_recording)
if(HONDADASH_WITH_SERIAL)
    find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS SerialPort)
    add_library(hondadash_serial src/transport/serial_transport.cpp)
    target_link_libraries(hondadash_serial PUBLIC hondadash_core Qt6::SerialPort)
    if(MSVC)
        target_compile_options(hondadash_serial PRIVATE /W4 /utf-8)
    endif()
endif()
foreach(target hondadash_core hondadash_recording hondadash_dlc hondadash_bridge hondadash_bench hondadash_acceptance)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
if(HONDADASH_BUILD_BRIDGE_CHECK)
    find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core)
    add_executable(HondaDashBridgeCheck src/tools/bridge_check_main.cpp)
    target_link_libraries(HondaDashBridgeCheck PRIVATE hondadash_acceptance Qt6::Core)
    target_compile_definitions(HondaDashBridgeCheck PRIVATE
        HD_DESKTOP_VERSION="${PROJECT_VERSION}" HD_DESKTOP_SHA="${HONDADASH_DESKTOP_SHA}")
    if(HONDADASH_WITH_SERIAL)
        target_link_libraries(HondaDashBridgeCheck PRIVATE hondadash_serial)
        target_compile_definitions(HondaDashBridgeCheck PRIVATE HONDADASH_WITH_SERIAL=1)
    endif()
    if(MSVC)
        target_compile_options(HondaDashBridgeCheck PRIVATE /W4 /utf-8)
    endif()
    install(TARGETS HondaDashBridgeCheck RUNTIME DESTINATION .)
endif()
if(HONDADASH_BUILD_GUI)
    find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Widgets)
    qt_add_executable(HondaDash src/ui/main.cpp src/ui/main_window.cpp src/ui/dashboard_widgets.cpp)
    target_link_libraries(HondaDash PRIVATE hondadash_recording hondadash_bench Qt6::Widgets)
    if(HONDADASH_WITH_SERIAL)
        target_link_libraries(HondaDash PRIVATE hondadash_serial)
        target_compile_definitions(HondaDash PRIVATE HONDADASH_WITH_SERIAL=1)
    endif()
    if(MSVC)
        target_compile_options(HondaDash PRIVATE /W4 /utf-8)
    endif()
    set_target_properties(HondaDash PROPERTIES WIN32_EXECUTABLE TRUE)
    install(TARGETS HondaDash RUNTIME DESTINATION .)
endif()
if(BUILD_TESTING)
    add_executable(one_wire_tests tests/one_wire_tests.cpp)
    target_link_libraries(one_wire_tests PRIVATE hondadash_one_wire nano_bridge)
    add_test(NAME one_wire COMMAND one_wire_tests)
    add_executable(bench_embedded_tests tests/bench_embedded_tests.cpp)
    target_link_libraries(bench_embedded_tests PRIVATE hondadash_core nano_bench)
    add_test(NAME bench_embedded COMMAND bench_embedded_tests)
    add_executable(bench_integration_tests tests/bench_integration_tests.cpp)
    target_link_libraries(bench_integration_tests PRIVATE hondadash_acceptance nano_bench)
    target_compile_definitions(bench_integration_tests PRIVATE HD_DESKTOP_VERSION="${PROJECT_VERSION}" HD_DESKTOP_SHA="${HONDADASH_DESKTOP_SHA}")
    if(MSVC)
        target_compile_options(bench_integration_tests PRIVATE /utf-8)
    endif()
    add_test(NAME bench_integration COMMAND bench_integration_tests --report-dir ${CMAKE_BINARY_DIR}/bench-integration-reports)
    set_tests_properties(bench_integration PROPERTIES TIMEOUT 120)
    add_executable(honda_dlc_tests tests/honda_dlc_tests.cpp)
    target_link_libraries(honda_dlc_tests PRIVATE hondadash_dlc hondadash_recording)
    if(MSVC)
        target_compile_options(honda_dlc_tests PRIVATE /W4 /utf-8)
    endif()
    add_test(NAME honda_dlc COMMAND honda_dlc_tests)
    foreach(name core session recording)
        add_executable(${name}_tests tests/${name}_tests.cpp)
        target_link_libraries(${name}_tests PRIVATE hondadash_recording)
        if(MSVC)
            target_compile_options(${name}_tests PRIVATE /utf-8)
        endif()
        add_test(NAME ${name} COMMAND ${name}_tests)
    endforeach()
    add_executable(embedded_tests tests/embedded_tests.cpp)
    target_link_libraries(embedded_tests PRIVATE hondadash_core nano_endpoint)
    add_test(NAME embedded COMMAND embedded_tests)
    add_executable(bridge_embedded_tests tests/bridge_embedded_tests.cpp)
    target_link_libraries(bridge_embedded_tests PRIVATE hondadash_core nano_bridge)
    add_test(NAME bridge_embedded COMMAND bridge_embedded_tests)
    add_executable(bridge_tests tests/bridge_tests.cpp)
    target_link_libraries(bridge_tests PRIVATE hondadash_bridge hondadash_recording nano_endpoint)
    if(MSVC)
        target_compile_options(bridge_tests PRIVATE /W4 /utf-8)
    endif()
    add_test(NAME bridge COMMAND bridge_tests)
    add_executable(polling_tests tests/polling_tests.cpp)
    target_link_libraries(polling_tests PRIVATE hondadash_bridge)
    if(MSVC)
        target_compile_options(polling_tests PRIVATE /W4 /utf-8)
    endif()
    add_test(NAME polling COMMAND polling_tests --report-dir ${CMAKE_BINARY_DIR}/timing-reports)
    add_executable(acceptance_tests tests/acceptance_tests.cpp)
    target_link_libraries(acceptance_tests PRIVATE hondadash_acceptance nano_endpoint)
    target_compile_definitions(acceptance_tests PRIVATE
        HD_DESKTOP_VERSION="${PROJECT_VERSION}" HD_DESKTOP_SHA="${HONDADASH_DESKTOP_SHA}")
    if(MSVC)
        target_compile_options(acceptance_tests PRIVATE /W4 /utf-8)
    endif()
    add_test(NAME acceptance COMMAND acceptance_tests --report-dir ${CMAKE_BINARY_DIR}/acceptance-reports)
    set_tests_properties(acceptance PROPERTIES TIMEOUT 60)
    if(HONDADASH_BUILD_BRIDGE_CHECK)
        add_test(NAME acceptance_native_cli COMMAND HondaDashBridgeCheck --backend native --report ${CMAKE_BINARY_DIR}/acceptance-native-cli)
        set_tests_properties(acceptance_native_cli PROPERTIES TIMEOUT 60)
    endif()
    add_executable(in_memory_tests tests/in_memory_tests.cpp)
    target_link_libraries(in_memory_tests PRIVATE hondadash_core)
    add_test(NAME in_memory COMMAND in_memory_tests)
    add_executable(session_embedded_tests tests/session_embedded_tests.cpp)
    target_link_libraries(session_embedded_tests PRIVATE hondadash_recording nano_endpoint)
    add_test(NAME session_embedded COMMAND session_embedded_tests)
    if(HONDADASH_WITH_SERIAL)
        add_executable(serial_tests tests/serial_tests.cpp)
        target_link_libraries(serial_tests PRIVATE hondadash_serial)
        add_test(NAME serial COMMAND serial_tests)
        if(UNIX AND NOT APPLE)
            add_executable(serial_pty_tests tests/serial_pty_tests.cpp)
            target_link_libraries(serial_pty_tests PRIVATE hondadash_serial nano_endpoint util)
            add_test(NAME serial_pty COMMAND serial_pty_tests)
            set_tests_properties(serial_pty PROPERTIES TIMEOUT 45)
            add_executable(bridge_pty_tests tests/bridge_pty_tests.cpp)
            target_link_libraries(bridge_pty_tests PRIVATE hondadash_bridge hondadash_serial util)
            add_test(NAME bridge_pty COMMAND bridge_pty_tests)
            set_tests_properties(bridge_pty PROPERTIES TIMEOUT 45)
            add_executable(acceptance_pty_tests tests/acceptance_pty_tests.cpp)
            target_link_libraries(acceptance_pty_tests PRIVATE hondadash_acceptance hondadash_serial util)
            target_compile_definitions(acceptance_pty_tests PRIVATE
                HD_DESKTOP_VERSION="${PROJECT_VERSION}" HD_DESKTOP_SHA="${HONDADASH_DESKTOP_SHA}")
            add_test(NAME acceptance_pty COMMAND acceptance_pty_tests --report-dir ${CMAKE_BINARY_DIR}/acceptance-pty-reports)
            set_tests_properties(acceptance_pty PROPERTIES TIMEOUT 60)
        endif()
    endif()
    if(HONDADASH_BUILD_GUI)
        add_test(NAME gui_smoke COMMAND HondaDash --smoke-test --report ${CMAKE_BINARY_DIR}/smoke-$<CONFIG>.json --screenshot ${CMAKE_BINARY_DIR}/smoke-$<CONFIG>.png)
        set_tests_properties(gui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 180)
    endif()
endif()
